# ESP-IDF component definition for tuya-iot-core-sdk.
#
# This file replaces the upstream standalone-project CMakeLists.txt so that
# ESP-IDF can build the SDK as a regular component. It was written to work
# with ESP-IDF v5.x (CMake >= 3.16, mbedtls 3.x).
#
# What is deliberately excluded:
#   platform/posix/mbedtls_sockets_wrapper.c  -- already in ESP-IDF's mbedtls component
#   platform/posix/network_wrapper.c           -- replaced by platform/esp32/network_wrapper.c
#                                                 (esp_tls based; no direct mbedtls 2.x API usage)
#   platform/posix/storage_wrapper.c           -- replaced by platform/esp32/storage_wrapper.c
#                                                 (NVS-backed, no filesystem required)
#   src/cipher_wrapper.c                       -- replaced by platform/esp32/cipher_wrapper.c
#                                                 (mbedtls/cipher.h removed in mbedtls 4.x; uses PSA AEAD)
#   libraries/mbedtls/**                        -- use ESP-IDF's mbedtls via REQUIRES

cmake_minimum_required(VERSION 3.16)

# Remove the Tuya SDK's bundled mbedtls before ESP-IDF's header scanner runs.
# ESP-IDF errors when the same header (e.g. mbedtls/cipher.h) physically exists
# in multiple component directories. We use ESP-IDF's built-in mbedtls component
# (via REQUIRES mbedtls below), so the bundled copy is not needed.
file(REMOVE_RECURSE "${CMAKE_CURRENT_SOURCE_DIR}/libraries/mbedtls")

idf_component_register(
    SRCS
        # Tuya link SDK core
        "src/tuyalink_core.c"
        # src/cipher_wrapper.c replaced by platform/esp32/cipher_wrapper.c below
        "src/iotdns.c"
        "src/tuya_endpoint.c"

        # coreJSON (embedded JSON parser)
        "libraries/coreJSON/source/core_json.c"

        # coreMQTT -- used internally for Tuya cloud connection
        # (separate from the ESP-MQTT component used for Home Assistant)
        "libraries/coreMQTT/source/core_mqtt.c"
        "libraries/coreMQTT/source/core_mqtt_state.c"
        "libraries/coreMQTT/source/core_mqtt_serializer.c"

        # coreHTTP -- used for OTA and Tuya API calls
        "libraries/coreHTTP/source/core_http_client.c"
        "libraries/coreHTTP/source/dependency/3rdparty/http_parser/http_parser.c"

        # Middleware: Tuya MQTT + HTTP wrappers around coreMQTT / coreHTTP
        "middleware/http_client_wrapper.c"
        "middleware/mqtt_client_wrapper.c"

        # Platform port (all ESP32-native replacements)
        # network_wrapper.c -- TLS via esp_tls (no direct mbedtls API)
        # cipher_wrapper.c  -- AEAD via PSA Crypto (mbedtls/cipher.h removed in mbedtls 4.x)
        # system_wrapper.c  -- clock_gettime/gettimeofday/nanosleep (available in ESP-IDF)
        # storage_wrapper.c -- NVS-backed ESP32 version
        "platform/esp32/network_wrapper.c"
        "platform/esp32/cipher_wrapper.c"
        "platform/posix/system_wrapper.c"
        "platform/esp32/storage_wrapper.c"

        # Utilities
        "utils/MultiTimer.c"
        "utils/base64.c"
        "utils/cJSON.c"
        "utils/log.c"

    INCLUDE_DIRS
        # platform/esp32 first: provides mbedtls/certs.h compatibility shim
        # (mbedtls 3.x used by ESP-IDF 5.x removed certs.h)
        "platform/esp32"
        "include"
        "interface"
        "utils"
        "libraries/coreJSON/source/include"
        "libraries/coreMQTT/source/include"
        "libraries/coreHTTP/source/include"
        "libraries/coreHTTP/source/dependency/3rdparty/http_parser"
        "middleware"
        "platform/posix"

    REQUIRES
        esp-tls
        mbedtls
        nvs_flash
        lwip
        esp_timer
)

# Per-file warning suppressions for third-party SDK files.
# These files have minor C89/C99-era issues that are not bugs in our usage,
# but -Werror in ESP-IDF 6.x / C23 mode promotes them to build failures.

# coreJSON: (1) uses `true` as an enum constant — illegal in C23 → force C11.
#           (2) char is unsigned on RISC-V, so `>= '\0'` is always true → -Wno-type-limits.
set_source_files_properties(
    "${CMAKE_CURRENT_SOURCE_DIR}/libraries/coreJSON/source/core_json.c"
    PROPERTIES COMPILE_FLAGS "-std=c11 -Wno-type-limits"
)

# tuyalink_core.c: (1) %d used for uint32_t (should be %lu/-PRIu32) → -Wno-format.
#                  (2) signed/unsigned pointer mismatch (char* vs uint8_t*) → -Wno-pointer-sign.
#                  (3) `return;` in non-void function (SDK bug, not on our code path) → -Wno-return-type.
#                  (4) unused local variable → -Wno-unused-variable.
set_source_files_properties(
    "${CMAKE_CURRENT_SOURCE_DIR}/src/tuyalink_core.c"
    PROPERTIES COMPILE_FLAGS "-Wno-format -Wno-pointer-sign -Wno-return-type -Wno-return-mismatch -Wno-unused-variable"
)
